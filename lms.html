async function loginStudent(){ const name = nameInput.value.trim(); const email = emailInput.value.trim().toLowerCase(); const password = passwordInput.value; msg.textContent = ""; msg.className = "msg"; if(!email){ msg.textContent = "❌ Email डालें"; msg.classList.add("error"); return; } if(!password || password.length < 6){ msg.textContent = "❌ Password कम से कम 6 characters का होना चाहिए"; msg.classList.add("error"); return; } loginBtn.disabled = true; loginBtn.textContent = "⏳ Login हो रहा है..."; try{ const loginData = await supabaseRequest( "/auth/v1/token?grant_type=password", { method: "POST", body: JSON.stringify({ email: email, password: password }) } ); if(!loginData.access_token){ throw new Error("Login token नहीं मिला"); } localStorage.setItem( "dc_access_token", loginData.access_token ); localStorage.setItem( "dc_user", JSON.stringify(loginData.user || {}) ); const studentName = name || loginData.user?.user_metadata?.name || loginData.user?.email?.split("@")[0] || "Student"; loginBox.classList.add("hidden"); dashboard.classList.remove("hidden"); welcome.textContent = studentName; msg.textContent = ""; await loadCourses(); }catch(error){ console.error("LOGIN ERROR:", error); msg.textContent = "❌ Login failed: " + (error.message || "Unknown error"); msg.classList.add("error"); }finally{ loginBtn.disabled = false; loginBtn.textContent = "🔐 Login"; } }
