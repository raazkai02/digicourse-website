async function loadCourses(){ courses.innerHTML="⏳ आपके courses loading..."; try{ const token = localStorage.getItem("dc_access_token"); if(!token){ throw new Error("Student login required"); } const user = JSON.parse( localStorage.getItem("dc_user") || "{}" ); if(!user.id){ throw new Error("Student session नहीं मिली"); } /* केवल student के खरीदे/enrolled courses */ const enrollments = await supabaseRequest( "/rest/v1/student_courses" + "?student_id=eq." + encodeURIComponent(user.id) + "&select=course_id" ); if(!enrollments || enrollments.length===0){ courses.innerHTML=`

📚 अभी कोई Course नहीं है

आपने अभी कोई course purchase नहीं किया है।

🛒 Course खरीदें
`; return; } const ids = enrollments .map(x => x.course_id) .filter(Boolean); if(ids.length===0){ courses.innerHTML= "
📚 कोई course नहीं मिला।
"; return; } const courseData = await supabaseRequest( "/rest/v1/courses" + "?id=in.(" + ids.map(encodeURIComponent).join(",") + ")" + "&select=id,title,description,price,original_price,offer_price" ); if(!courseData || courseData.length===0){ courses.innerHTML=`
📚 आपका purchased course अभी उपलब्ध नहीं है।
`; return; } courses.innerHTML = courseData.map(course=>`

📚 ${escapeHTML(course.title)}

${escapeHTML(course.description || "")}

₹${course.offer_price ?? course.price ?? 0}
✅ Purchased
`).join(""); } catch(error){ console.error(error); courses.innerHTML=`
❌ Courses load नहीं हो पाए।

${escapeHTML(error.message)}
`; } }
