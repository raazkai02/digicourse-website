async function loginAdmin(){ const email = document.getElementById("email").value.trim().toLowerCase(); const password = document.getElementById("pass").value; const msg = document.getElementById("msg"); msg.textContent = ""; if(!email || !password){ msg.textContent = "❌ Email और Password डालें"; return; } msg.textContent = "⏳ Login हो रहा है..."; try{ /* STEP 1: SUPABASE AUTH LOGIN */ const data = await api( "/auth/v1/token?grant_type=password", { method:"POST", body:JSON.stringify({ email:email, password:password }) } ); if(!data.access_token){ throw new Error("Login token नहीं मिला"); } /* TOKEN SAVE */ T = data.access_token; localStorage.setItem( "admin_token", T ); localStorage.setItem( "admin_user_id", data.user.id ); /* STEP 2: ADMIN TABLE में EMAIL से CHECK */ const admins = await api( "/rest/v1/admins?email=eq." + encodeURIComponent(email) + "&select=*" ); if(!admins || admins.length === 0){ T = ""; localStorage.removeItem("admin_token"); localStorage.removeItem("admin_user_id"); throw new Error( "यह Gmail admins table में registered नहीं है" ); } /* STEP 3: ADMIN LOGIN SUCCESS */ msg.textContent = "✅ Admin Login Successfully"; showPanel(); load(); } catch(error){ console.error(error); T = ""; localStorage.removeItem("admin_token"); localStorage.removeItem("admin_user_id"); msg.textContent = "❌ " + error.message; } }
