async function loginAdmin(){ const email = document.getElementById("email").value.trim().toLowerCase(); const password = document.getElementById("pass").value; const msg = document.getElementById("msg"); msg.textContent = ""; if(!email || !password){ msg.textContent = "❌ Email और Password डालें"; return; } msg.textContent = "⏳ Admin Login हो रहा है..."; try{ /* AUTH LOGIN */ const data = await api( "/auth/v1/token?grant_type=password", { method:"POST", body:JSON.stringify({ email:email, password:password }) } ); if(!data.access_token){ throw new Error("Login token नहीं मिला"); } T = data.access_token; /* ADMIN CHECK — केवल ID से */ const admins = await api( "/rest/v1/admins?id=eq." + encodeURIComponent(data.user.id) + "&select=id,name" ); if(!admins || admins.length === 0){ T = ""; throw new Error( "❌ यह Account Admin के रूप में registered नहीं है" ); } /* SAVE SESSION */ localStorage.setItem( "admin_token", T ); localStorage.setItem( "admin_user_id", data.user.id ); /* SUCCESS */ msg.textContent = "✅ Admin Login Successfully"; showPanel(); load(); } catch(error){ console.error("ADMIN LOGIN:",error); T = ""; localStorage.removeItem("admin_token"); localStorage.removeItem("admin_user_id"); msg.textContent = "❌ " + error.message; } }
